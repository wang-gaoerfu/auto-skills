#### 2.6 AI生成流程（可复用90%）
```javascript
// 参考项目的AI生成流程
async function generateContent(type, params) {
    // 1. 构建提示词
    let prompt = PROMPT_TEMPLATES[type];
    prompt = replaceVariables(prompt, params);

    // 2. 调用AI
    const result = await callAI(prompt, params.model);

    // 3. 处理结果
    if (params.stream) {
        return handleStreamResult(result);
    } else {
        return handleResult(result);
    }
}

// 流式输出处理
async function* handleStreamResult(response) {
    const reader = response.body.getReader();
    while (true) {
        const { done, value } = await reader.read();
        if (done) break;
        const text = decoder.decode(value);
        const lines = text.split('\n').filter(line => line.trim());
        for (const line of lines) {
            if (line.startsWith('data: ')) {
                const data = JSON.parse(line.slice(6));
                yield data.content;
            }
        }
    }
}
```

**复用度：90%** - 流程完全可用，只需适配新的API
