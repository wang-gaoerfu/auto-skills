```prisma
model ScriptProject {
  id                  String    @id @default(cuid())
  userId              String
  title               String
  description         String?
  coverImage          String?
  sourceType          String    @default("ORIGINAL")
  sourceProjectId     String?
  sourceNovelTitle    String?
  genre               String?
  status              String    @default("draft")
  subStatus           String?
  progress            Int       @default(0)
  generationLock      String?
  lockAcquiredAt      DateTime?
  settings            Json?
  totalTokens         Int       @default(0)
  totalShots          Int       @default(0)
  totalDuration       Int       @default(0)
  totalScenes         Int       @default(0)
  lastExportAt        DateTime?
  exportCount         Int       @default(0)
  createdAt           DateTime  @default(now())
  updatedAt           DateTime  @updatedAt

  user                User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  sourceProject       Project?  @relation(fields: [sourceProjectId], references: [id], onDelete: SetNull)
  sources             ScriptSource[]
  characters          ScriptCharacter[]
  scenes              ScriptScene[]
  generationTasks     ScriptGenerationTask[]
  exports             ScriptExport[]
  costLogs            ScriptCostLog[]

  @@index([userId])
  @@index([status])
  @@index([createdAt])
  @@map("script_projects")
}
```
