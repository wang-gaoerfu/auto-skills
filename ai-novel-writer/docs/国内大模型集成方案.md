### 3.3 故障转移机制

```python
class ModelManager:
    """模型管理器，支持自动切换"""

    def __init__(self):
        self.models = {
            "deepseek": DeepSeekClient(),
            "kimi": KimiClient(),
            "qwen": QwenClient(),
            "glm": GLMClient()
        }
        self.model_health = {
            "deepseek": True,
            "kimi": True,
            "qwen": True,
            "glm": True
        }

    async def generate(self, prompt: str, task_type: str):
        """智能生成，自动故障转移"""

        # 获取该任务的首选模型列表
        model_list = MODEL_STRATEGY[task_type]

        # 尝试主模型
        primary = model_list["primary"]
        if self.model_health[primary]:
            try:
                return await self.models[primary].generate(prompt)
            except Exception as e:
                # 标记模型不可用
                self.model_health[primary] = False
                logger.error(f"{primary} 失败: {e}")

        # 尝试备用模型
        fallback = model_list.get("fallback")
        if fallback and self.model_health[fallback]:
            return await self.models[fallback].generate(prompt)

        # 都不行，用默认
        return await self.models["deepseek"].generate(prompt)

    async def health_check(self):
        """定期健康检查"""
        for model_name, client in self.models.items():
            try:
                await client.ping()
                self.model_health[model_name] = True
            except:
                self.model_health[model_name] = False
```

## 五、统一接口封装

### 5.1 统一调用接口

```python
from abc import ABC, abstractmethod
from typing import AsyncIterator

class BaseAIModel(ABC):
    """所有模型的基础接口"""

    @abstractmethod
    async def generate(
        self,
        prompt: str,
        temperature: float = 0.7,
        max_tokens: int = 2000
    ) -> str:
        """生成文本（非流式）"""
        pass

    @abstractmethod
    async def generate_stream(
        self,
        prompt: str,
        temperature: float = 0.7,
        max_tokens: int = 2000
    ) -> AsyncIterator[str]:
        """生成文本（流式）"""
        pass

    @abstractmethod
    async def count_tokens(self, text: str) -> int:
        """计算token数"""
        pass

# 使用示例
ai_model = ModelManager.get_model("deepseek")

# 非流式
result = await ai_model.generate("写一个小说开头")

# 流式
async for chunk in ai_model.generate_stream("写一个小说开头"):
    print(chunk, end="")
```

### 6.1 使用监控

```python
class ModelMonitor:
    """模型使用监控"""

    async def track_usage(
        self,
        user_id: int,
        model: str,
        tokens: int,
        cost: float
    ):
        """记录使用情况"""
        await redis.hincrby(f"user:{user_id}:stats", "total_tokens", tokens)
        await redis.hincrbyfloat(f"user:{user_id}:stats", "total_cost", cost)
        await redis.hincrby(f"model:{model}:stats", "total_tokens", tokens)

    async def get_daily_report(self):
        """生成日报"""
        return {
            "total_tokens": await redis.get("stats:daily_tokens"),
            "total_cost": await redis.get("stats:daily_cost"),
            "model_usage": {
                model: await redis.hgetall(f"model:{model}:stats")
                for model in ["deepseek", "kimi", "qwen", "glm"]
            }
        }
```
