## 一、功能概述

### 1.1 功能定位

为 AI 小说创作能手平台新增**漫剧剧本生成**模块，支持将小说内容（自有/外部）转换为专业的分镜剧本格式，用于后续 AI 漫剧/短剧制作。

### 1.2 核心价值

| 价值点 | 说明 |
|--------|------|
| **内容变现** | 小说作者可将作品转化为漫剧剧本，拓展变现渠道 |
| **降低门槛** | 无需专业编剧知识，AI 自动生成分镜脚本 |
| **解耦设计** | 剧本系统独立于小说系统，互不影响 |
| **多来源支持** | 支持自有项目、外部小说、从零创作 |

### 1.3 与现有系统的关系

```
┌─────────────────────────────────────────────────────────────┐
│                    AI 小说创作能手                            │
├─────────────────────────────────────────────────────────────┤

│  │   小说系统       │   →→→   │   剧本系统       │           │
│  │  (现有功能)      │  导入   │  (新增功能)      │           │
│  │                 │         │                 │           │
│  │  - 项目管理      │         │  - 剧本项目管理  │           │
```

## 二、需求分析

### 2.1 用户故事

| 角色 | 需求 | 价值 |
|------|------|------|
| 小说作者 | 我想把我的小说转成漫剧剧本 | 拓展作品形式，增加收入 |
| 短剧创作者 | 我想导入别人的小说生成剧本 | 快速获取素材，提高效率 |
| 内容运营 | 我想批量生成剧本用于制作 | 降低内容生产成本 |
| 编剧新手 | 我想学习专业剧本格式 | AI 生成的规范格式可作为参考 |

### 2.4 会员机制设计（独立权限）

> **设计原则**：剧本系统采用**独立会员体系**，与小说系统解耦，用户可单独购买剧本权益。

#### 2.4.1 独立权限设计理由

| 考量 | 说明 |
|------|------|
| 用户画像不同 | 60%只写小说，25%只做剧本，15%两者都需要 |
| 成本结构不同 | 剧本AI调用更复杂，但频次可能更低 |
| 产品定位不同 | 小说=长期创作工具，剧本=项目制转换工具 |
| 灵活定价 | 可根据剧本系统成本独立定价 |

```prisma
model ScriptMembership {
  id              String    @id @default(cuid())
  userId          String    @unique
  plan            String    @default("FREE") // FREE, VIP, PRO
  status          String    @default("ACTIVE") // ACTIVE, EXPIRED, CANCELLED
  expiresAt       DateTime?
  createdAt       DateTime  @default(now())
  updatedAt       DateTime  @updatedAt

  // 使用统计
  projectCount    Int       @default(0)
  dailyGenerations Int      @default(0)
  lastGenerationDate DateTime?

  // 兑换码激活（可选）
  redemptionCodeId String?
  redemptionCode   RedemptionCode? @relation(fields: [redemptionCodeId], references: [id])

  user            User      @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@index([userId])
  @@index([status])
  @@map("script_memberships")
}

  id            String   @id @default(cuid())

  type          String   @default("NOVEL") // NOVEL, SCRIPT, COMBO

  usedCount     Int      @default(0)

  expiresAt     DateTime?

  createdAt     DateTime @default(now())

  novelMemberships  Membership[]

  @@index([code])
```

```typescript
// 剧本系统配额检查
export async function checkScriptQuota(userId: string) {
  const membership = await prisma.scriptMembership.findUnique({
    where: { userId }
  }) || { plan: 'FREE', projectCount: 0, dailyGenerations: 0 }

  const limits = {
    FREE: {
      maxProjects: 1,
      maxChaptersPerProject: 3,
      dailyGenerations: 3,
      templates: ['basic'],
      exportFormats: ['json', 'md'],
      hasWatermark: true,
      hasApi: false,
      hasBatch: false
    },
    VIP: {
      maxProjects: 10,
      maxChaptersPerProject: Infinity,
      dailyGenerations: 50,
      templates: ['basic', 'advanced', 'cinematic'],
      exportFormats: ['json', 'md', 'pdf'],
      hasWatermark: false,
      hasApi: false,
      hasBatch: false
    },
    PRO: {
      maxProjects: Infinity,
      maxChaptersPerProject: Infinity,
      dailyGenerations: Infinity,
      templates: ['basic', 'advanced', 'cinematic', 'custom'],
      exportFormats: ['json', 'md', 'pdf', 'excel'],
      hasWatermark: false,
      hasApi: true,
      hasBatch: true
    }
  }

  return {
    plan: membership.plan,
    limits: limits[membership.plan],
    usage: {
      projectCount: membership.projectCount,
      dailyGenerations: membership.dailyGenerations,
      remainingGenerations: limits[membership.plan].dailyGenerations - membership.dailyGenerations
    }
  }
}

// 每日配额重置（定时任务）
export async function resetDailyQuota() {
  await prisma.scriptMembership.updateMany({
    where: { lastGenerationDate: { lt: today() } },
    data: { dailyGenerations: 0 }
  })
}
```

## 三、剧本格式设计

### 3.1 分镜剧本格式规范

```yaml
# 剧本元数据
meta:
  title: "剧本标题"
  source: "来源小说名称"
  genre: "都市爱情"          # 题材类型
  totalScenes: 12           # 场景总数
  estimatedDuration: "5分钟" # 预估时长
  characters:               # 角色列表
    - id: "char_001"
      name: "小雨"
      description: "25岁女性，黑色长发，白色连衣裙"
    - id: "char_002"
      name: "林远"
      description: "28岁男性，短发，西装"

# 场景列表
scenes:
  - id: "scene_001"
    sceneNumber: 1
    title: "咖啡馆门口"
    timeOfDay: "日"          # 日/夜/黄昏/黎明
    location: "外"           # 内/外
    description: "繁华街道边的咖啡馆门口，人来人往"
    mood: "期待、紧张"       # 场景氛围
    shots:                   # 镜头列表
      - id: "shot_001_001"
        shotNumber: "1-1"
        shotType: "中景"     # 特写/近景/中景/全景/远景
        angle: "平视"        # 平视/仰视/俯视/斜角
        duration: 3          # 预估秒数

        # 画面描述（用于 AI 生图）
        visual:
          subject: "小雨站在咖啡馆门口"
          characters:
            - characterId: "char_001"
              action: "来回踱步"
              expression: "紧张、期待"
              position: "画面中央"
          environment: "咖啡馆门口，街道背景，行人模糊"
          lighting: "自然光，阳光明媚"

        # 音频描述
        audio:
          dialogue:
            - character: "小雨（内心）"
              line: "他会不会来..."
              type: "旁白"    # 对白/旁白/内心独白
          soundEffects:
            - "街道嘈杂声"
            - "手机震动声"
          backgroundMusic: "轻柔的钢琴曲"

        # 镜头运动
        camera:
          movement: "固定"    # 固定/推/拉/摇/移/跟
          transition: "切"    # 切/淡入/淡出/叠化

      - id: "shot_001_002"
        shotNumber: "1-2"
        shotType: "特写"
        visual:
          subject: "手机屏幕"
          environment: "显示来电：林远"
        audio:
          soundEffects:
            - "手机震动声"
        duration: 2

  - id: "scene_002"
    sceneNumber: 2
    title: "咖啡馆内"
    # ... 更多场景
```

```typescript
const STATE_TRANSITIONS = {

    canTransitionTo: ['generating'],
    allowedActions: ['edit', 'delete', 'start_generate']
  },
  generating: {
    canTransitionTo: ['completed', 'error', 'draft'],
    allowedActions: ['cancel', 'view_progress'],

  completed: {

function canTransition(currentStatus: string, targetStatus: string): boolean {
  return STATE_TRANSITIONS[currentStatus]?.canTransitionTo.includes(targetStatus) ?? false
}
```

### 5.7 编码检测与转换

```typescript
// src/lib/script/encoding.ts
import * as iconv from 'iconv-lite';

/**
 * 检测文件编码并转换为 UTF-8
 */
async function detectAndConvertEncoding(buffer: Buffer): Promise<string> {
  // 1. 尝试 UTF-8
  try {
    const text = buffer.toString('utf-8');
    if (!text.includes('�')) {
      return text;
    }
  } catch {}

  // 2. 尝试 GBK
  try {
    const text = iconv.decode(buffer, 'gbk');
    if (isValidText(text)) {
      return text;
    }
  } catch {}

  // 3. 尝试 GB2312
  try {
    const text = iconv.decode(buffer, 'gb2312');
    if (isValidText(text)) {
      return text;
    }
  } catch {}

  // 4. 无法识别编码
  throw new Error('UNSUPPORTED_ENCODING');
}

/**
 * 检查文本是否有效（无乱码）
 */
function isValidText(text: string): boolean {
  // 检查是否包含过多替换字符
  const replacementCount = (text.match(/�/g) || []).length;
  const threshold = text.length * 0.01; // 1% 阈值
  return replacementCount < threshold;
}
```

### 5.13 边界场景处理

#### 5.13.1 输入边界

| 场景 | 检测条件 | 处理方式 |
|------|---------|---------|
| 章节内容为空 | `content.length === 0` | 提示"章节内容为空，请补充内容" |
| 章节过短 | `content.length < 100` | 提示"章节内容过短，建议至少100字" |
| 章节过长 | `content.length > 100000` | 自动分段处理，或提示拆分 |
| 总字数超限 | `totalWords > 500000` | 提示"内容过长，建议分批导入" |
| 文件编码异常 | 无法识别编码 | 提示"编码格式不支持，请转换为 UTF-8" |

#### 5.13.2 生成边界

| 场景 | 检测条件 | 处理方式 |
|------|---------|---------|
| AI 调用超时 | `timeout > 60s` | 自动重试（最多3次），提示用户 |
| AI 返回空 | `result === null` | 重试，记录错误日志 |
| AI 返回格式错误 | JSON 解析失败 | 重试，使用更严格的提示词 |
| 内容违规 | `content_filter` | 拒绝生成，提示修改内容 |
| 并发生成冲突 | 同项目多任务 | 加锁，排队处理 |

#### 5.13.3 编辑边界

| 场景 | 检测条件 | 处理方式 |
|------|---------|---------|
| 角色重名 | `name === existingName` | 自动添加序号 "角色名(2)" |
| 场景重名 | `title === existingTitle` | 自动添加序号 "场景名(2)" |
| 镜头顺序冲突 | `order` 重复 | 自动重新排序 |
| 并发编辑冲突 | 乐观锁失败 | 提示刷新后重试 |

#### 5.13.4 导出边界

| 场景 | 检测条件 | 处理方式 |
|------|---------|---------|
| 项目无内容 | `shotCount === 0` | 提示"暂无内容可导出" |
| 导出文件过大 | `fileSize > 50MB` | 分批导出或压缩 |
| PDF 生成失败 | 渲染错误 | 提供 Markdown 替代下载 |
| 网络中断 | 下载中断 | 提供重试按钮 |

#### 8.2.1 流式生成实现

```typescript
// /api/scripts/[id]/generate/route.ts
export async function POST(request: Request) {
  const encoder = new TextEncoder()

  const stream = new ReadableStream({
    async start(controller) {
      // 1. 获取章节内容
      const sources = await getScriptSources(scriptProjectId)

      // 2. 逐章生成
      for (const source of sources) {
        // 发送开始信号
        controller.enqueue(encoder.encode(`data: ${JSON.stringify({
          type: 'start',
          chapterOrder: source.order,
          chapterTitle: source.chapterTitle
        })}\n\n`))

        // AI 生成（流式）
        for await (const chunk of generateScriptStream(source.content, characters, scenes)) {
          controller.enqueue(encoder.encode(`data: ${JSON.stringify({
            type: 'chunk',
            content: chunk.content
          })}\n\n`))
        }

        // 发送完成信号
        controller.enqueue(encoder.encode(`data: ${JSON.stringify({
          type: 'complete',
          chapterOrder: source.order
        })}\n\n`))
      }

      controller.close()
    }
  })

  return new Response(stream, {
    headers: { 'Content-Type': 'text/event-stream' }
  })
}
```

```typescript
async function processLargeText(text: string) {
  const CHUNK_SIZE = 10000 // 每块 10000 字符

  if (text.length <= CHUNK_SIZE) {
    return await analyzeWithAI(text)
  }

  // 分块分析后合并
  const chunks = splitIntoChunks(text, CHUNK_SIZE)

}
```

#### 11.3.2 用户自带 API Key 模式

```typescript
// 用户 API Key 配置
model UserApiKeys {
  id              String  @id @default(cuid())
  userId          String  @unique

  // 图片生成
  siliconflowKey  String? // 加密存储
  midjourneyKey   String?
  stabilityKey    String?

  // 视频生成
  runwayKey       String?
  pikaKey         String?
  klingKey        String?

  // 配音
  minimaxKey      String?
  aliyunVoiceKey  String?

  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  @@map("user_api_keys")
}

// 生成时优先使用用户 Key
async function generateImage(params: ImageGenParams) {
  const userKeys = await getUserApiKeys(params.userId)

  // 优先级：用户 Key > 平台 Key
  if (userKeys.siliconflowKey) {
    return await siliconflowGenerate(params, userKeys.siliconflowKey)
  }

  // 回退到平台配额
  if (await checkPlatformQuota(params.userId)) {
    return await platformGenerate(params)
  }

  throw new Error('配额已用尽，请配置您的 API Key 或升级会员')
}
```

```typescript
你是一位专业的剧本审核编辑。请审核以下分镜内容：

1. 剧情连贯性：是否有逻辑漏洞
```
