### 4.1.3 验证码存储（Redis）

```typescript
// 使用 Redis 存储验证码（开发环境可用内存存储）
// lib/verification.ts

import { redis } from './redis'  // 或使用内存存储

const VERIFICATION_CODE_PREFIX = 'verification_code:'
const RESEND_LIMIT_PREFIX = 'resend_limit:'

// 存储验证码
export async function saveVerificationCode(email: string, code: string) {
  const key = VERIFICATION_CODE_PREFIX + email
  const expireSeconds = Number(process.env.VERIFICATION_CODE_EXPIRE_MINUTES) * 60

  await redis.set(key, code, 'EX', expireSeconds)
}

// 验证验证码
export async function verifyCode(email: string, code: string): Promise<boolean> {
  const key = VERIFICATION_CODE_PREFIX + email
  const savedCode = await redis.get(key)

  if (savedCode === code) {
    await redis.del(key)  // 验证成功后删除
    return true
  }
  return false
}

// 检查是否可以重发验证码
export async function canResendCode(email: string): Promise<boolean> {
  const key = RESEND_LIMIT_PREFIX + email
  const exists = await redis.exists(key)
  return !exists
}

// 设置重发限制
export async function setResendLimit(email: string) {
  const key = RESEND_LIMIT_PREFIX + email
  const interval = Number(process.env.VERIFICATION_CODE_RESEND_INTERVAL)
  await redis.set(key, '1', 'EX', interval)
}

// 清除重发限制
export async function clearResendLimit(email: string) {
  const key = RESEND_LIMIT_PREFIX + email
  await redis.del(key)
}
```

### 5.3 会员激活流程

**方式一：兑码激活（推荐）**

```typescript

- 适合批量发放兑换码场景
```

### 5.3.1 兑换码系统设计

```typescript
// 兑换码数据模型
model RedemptionCode {
  id            String   @id @default(cuid())
  code          String   @unique  // 兑换码：VIP-2024-XXXXXX
  plan          String            // VIP, PRO
  duration      Int               // 有效期（天）
  maxUses       Int               // 最大使用次数
  usedCount     Int      @default(0)  // 已使用次数
  status        String   @default("ACTIVE")  // ACTIVE, DISABLED, EXPIRED
  expiresAt     DateTime?          // 兑换码过期时间
  createdBy     String?           // 创建人ID
  createdAt     DateTime  @default(now())

  @@index([code])
  @@index([status])
}

// 兑换码格式
// VIP-2024-{6位随机码}  -> VIP会员30天
// PRO-2024-{6位随机码}  -> PRO会员30天
// VIP-2025-{6位随机码}  -> VIP会员365天
// PRO-2025-{6位随机码}  -> PRO会员365天

// 生成兑换码
function generateCode(plan: string, duration: number): string {
  const year = duration >= 365 ? '2025' : '2024'
  const random = Math.random().toString(36).substring(2, 8).toUpperCase()
  return `${plan}-${year}-${random}`
}

// 示例：
// VIP-2024-ABC123 -> VIP会员30天
// PRO-2025-XYZ789 -> PRO会员365天
```

### 6.3 向量检索流程

```typescript
// AI生成时自动检索相关知识
async function getRelevantKnowledge(
  projectId: string,
  currentContent: string
) {
  // 1. 向量检索相关人物（最多3个）
  const characters = await chroma.query({
    queryTexts: [currentContent],
    nResults: 3,
    where: { projectId, type: 'character' }
  })

  // 2. 向量检索相关世界观（最多2个）
  const world = await chroma.query({
    queryTexts: [currentContent],
    nResults: 2,
    where: { projectId, type: 'world' }
  })

  // 3. 向量检索相关章节（保持连贯性）
  const chapters = await chroma.query({
    queryTexts: [currentContent],
    nResults: 2,
    where: { projectId, type: 'chapter' }
  })

  return { characters, world, chapters }
}
```

## 十四、邮箱服务配置说明

### 14.1 163邮箱授权码获取步骤

1. 登录163邮箱网页版
2. 点击"设置" → "POP3/SMTP/IMAP"
3. 开启"SMTP服务"
4. 点击"新增授权码"
5. 通过短信验证后获取授权码
6. 将授权码填入 `MAIL_PASSWORD` 配置项
